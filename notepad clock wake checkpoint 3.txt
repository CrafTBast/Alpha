                localStorage.removeItem(`notepad_template_workspace_${oldId}`);

                    workspacesList: workspacesList,
                    contents: {},
                    templates: {}
                };
                
                workspacesList.forEach(wsId => {

                    compiledProfileData.templates[wsId] = localStorage.getItem(`notepad_template_workspace_${wsId}`) || "";
                });
                
                const serializedProfileString = JSON.stringify(compiledProfileData, null, 4);
                const blobConduit = new Blob([serializedProfileString], { type: "text/plain;charset=utf-8" });
                const downloadLinkAnchor = document.createElement("a");
                
                downloadLinkAnchor.href = URL.createObjectURL(blobConduit);
                downloadLinkAnchor.download = `workspace_master_setup_profile.txt`;
                document.body.appendChild(downloadLinkAnchor);
                downloadLinkAnchor.click();
                document.body.removeChild(downloadLinkAnchor);
            } catch (err) {
                alert("Export error occurred while packing your workstation settings profiles.");
            }
            textarea.focus();
        }
        
        // 📤 ACTION B: IMPORT AND OVERWRITE UNPACK PASS
        else if (chosenAction === "import") {
            qs("#ws-profile-file-picker").click();
        }
    };

    // 🔄 PICKER INTERCEPTOR MECHANISM: Reads and unpacks your pre-loaded layout rules instantly
    qs("#ws-profile-file-picker").onchange = function(e) {
        const selectedFile = this.files;
        if (!selectedFile || selectedFile.length === 0) return;

        const fileReaderConduit = new FileReader();
        fileReaderConduit.onload = function(event) {
            try {
                const parsedProfileMap = JSON.parse(event.target.result);
                
                if (!parsedProfileMap.workspacesList || !parsedProfileMap.contents || !parsedProfileMap.templates) {
                    throw new Error("Invalid workspace blueprint configuration layout metrics.");
                }
                
                if (!confirm("Are you sure you want to overwrite your active layout with this imported Master Setup profile? All existing note frames will be updated instantly.")) {
                    return;
                }

                localStorage.setItem("notepad_workspaces_list", JSON.stringify(parsedProfileMap.workspacesList));
                localStorage.setItem("notepad_active_workspace", parsedProfileMap.activeWorkspace || "1");

                parsedProfileMap.workspacesList.forEach(wsId => {
                    localStorage.setItem(`notepad_content_workspace_${wsId}`, parsedProfileMap.contents[wsId] || "");
                    localStorage.setItem(`notepad_template_workspace_${wsId}`, parsedProfileMap.templates[wsId] || "");
                });

                workspacesList = parsedProfileMap.workspacesList;
                activeWorkspace = parsedProfileMap.activeWorkspace || "1";

                renderWorkspaces();
                switchWorkspace(activeWorkspace);
                
                alert("Master Setup Profile loaded successfully! Your workspace default layouts have been completely restored.");
            } catch (err) {
                alert("Failed to unpack setup profile file. Ensure it is a valid workspace configuration file.");
            }
        };
        fileReaderConduit.readAsText(selectedFile[0]);
        this.value = ""; // Clear file anchor so it can trigger consecutively later
        textarea.focus();
    };

        if (e.key === " " || e.keyCode === 32) {
            if (!isOpen && document.activeElement.tagName !== 'INPUT' && document.activeElement.tagName !== 'TEXTAREA') {
                e.preventDefault();
                e.stopPropagation();
                openNotepad();
            }
            return;
        }

        // Handle Escape Key Traversal Paths
        if (e.key === "Escape" || e.keyCode === 27) {
            if (!isOpen) {
                e.preventDefault();
                e.stopPropagation();
                openNotepad();
            } else {
                // 🚀 DUAL-PRESS LOCK: Must hit Escape twice within 400ms to close
                if (currentTime - lastEscapeTime < 400) {
                    e.preventDefault();
                    e.stopPropagation();
                    closeNotepad();
                } else {
                    lastEscapeTime = currentTime;
                }
            }
            return;
        }

        // Handle Insert Key Traversal Paths
        if (e.key === "Insert" || e.keyCode === 45) {
            if (!isOpen) {
                e.preventDefault();
                e.stopPropagation();
                openNotepad();
            } else {
                // 🚀 DUAL-PRESS LOCK: Must hit Insert twice within 400ms to close
                if (currentTime - lastInsertTime < 400) {
                    e.preventDefault();
                    e.stopPropagation();
                    closeNotepad();
                } else {
                    lastInsertTime = currentTime;
                }
            }
            return;
        }
    }, true);

    document.addEventListener("keyup", function(e) {
        // Safely clear keys from our tracking map on hardware release
        delete keysPressedMap[e.key];
    }, true);

    // ==========================================================
    // 🖱️ UPDATED GLOBAL MOUSE LISTENERS: CHORD STRICT PAUSE ONLY
    // ==========================================================
    document.addEventListener("mousedown", function(e) {

/**
 * Injects a bottom-centered 15px diamond button that toggles the PC Wake Lock.
 * Remembers its active/inactive state across page refreshes using localStorage.
 */
function initializeWakeLockButton() {
    const oldHost = document.getElementById('wake-toggle-host');
    if (oldHost) oldHost.remove();

    const host = document.createElement('div');
    host.id = 'wake-toggle-host';
    document.documentElement.appendChild(host);

    const shadow = host.attachShadow({ mode: 'open' });

    // 1. Injected styles preset to 15px x 15px
    shadow.innerHTML = `
        <style>
            :host {
                position: fixed !important;
                bottom: 25px !important; /* Adjusted for the smaller 15px size */
                left: 50% !important;
                transform: translateX(-50%) !important;
                z-index: 2147483647 !important;
                isolation: isolate !important;
                pointer-events: auto !important;
                visibility: visible !important;
                display: block !important;
                opacity: 1 !important;
            }
            #wake-toggle-btn {
                display: block !important;
                width: 15px !important;  /* Your updated size */
                height: 15px !important; /* Your updated size */
                background-color: #ef4444 !important; /* Red (Inactive) */
                border: 1.5px solid #ffffff !important; /* Thinner border for small size */
                box-shadow: 0 2px 6px rgba(0, 0, 0, 0.5) !important;
                cursor: pointer !important;
                transform: rotate(45deg) !important;
                padding: 0 !important;
                box-sizing: border-box !important;
                transition: background-color 0.3s ease, transform 0.2s ease !important;
            }
            #wake-toggle-btn:hover {
                transform: rotate(45deg) scale(1.2) !important;
            }
            #wake-toggle-btn.active {
                background-color: #22c55e !important; /* Green (Active) */
                box-shadow: 0 0 8px #22c55e, 0 2px 6px rgba(0, 0, 0, 0.5) !important;
            }
        </style>
        <button id="wake-toggle-btn" aria-label="Toggle Wake Lock"></button>
    `;

    const btn = shadow.getElementById('wake-toggle-btn');
    let wakeLock = null;

    async function requestWakeLock() {
        try {
            wakeLock = await navigator.wakeLock.request('screen');
            btn.classList.add('active');
            localStorage.setItem('pcWakeLockActive', 'true'); // Save state
            
            wakeLock.addEventListener('release', () => {
                if (wakeLock !== null) handleRelease();
            });
        } catch (err) {
            handleRelease();
        }
    }

    function handleRelease() {
        wakeLock = null;
        btn.classList.remove('active');
        localStorage.setItem('pcWakeLockActive', 'false'); // Save state
    }

    // Toggle click action
    btn.addEventListener('click', async () => {
        if (!('wakeLock' in navigator)) {
            console.warn('Wake Lock API not supported.');
            return;
        }

        if (wakeLock === null) {
            await requestWakeLock();
        } else {
            await wakeLock.release();
            handleRelease();
        }
    });

    // Re-engage if user tabs away and returns
    document.addEventListener('visibilitychange', async () => {
        if (document.visibilityState === 'visible' && localStorage.getItem('pcWakeLockActive') === 'true') {
            await requestWakeLock();
        }
    });

    // 2. AUTO-RESTORE STATE ON REFRESH
    // Checks if it was active before the reload and re-activates it
    if (localStorage.getItem('pcWakeLockActive') === 'true') {
        // A tiny timeout ensures the browser window is fully loaded to accept the lock
        setTimeout(async () => {
            await requestWakeLock();
        }, 100);
    }
}
