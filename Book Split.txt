<script>
let splitFilesMemory = [];

function processTextFile() {
    const fileInput = document.getElementById('fileInput');
    const keyword = document.getElementById('keywordInput').value;
    const resultsSection = document.getElementById('resultsSection');
    const chapterList = document.getElementById('chapterList');

    if (!fileInput.files.length) {
        alert('Please upload a text file first.');
        return;
    }
    if (!keyword) {
        alert('Please enter a split keyword.');
        return;
    }

    const file = fileInput.files[0];
    const reader = new FileReader();

    reader.onload = function(e) {
        const fullText = e.target.result;
        const segments = fullText.split(keyword);
        
        if (segments.length <= 1) {
            alert('Keyword not found in the text. No splits made.');
            resultsSection.style.display = 'none';
            return;
        }

        splitFilesMemory = [];
        chapterList.innerHTML = '';
        let fileCount = 0;

        segments.forEach((segment, index) => {
            // Skip the first segment if it's completely empty space
            if (index === 0 && segment.trim() === "") return;

            // Put the keyword back at the beginning of the text block
            const fileContent = (index === 0) ? segment : keyword + segment;
            fileCount++;
            
            // --- NEW FILENAME LOGIC ---
            let fileName = "";
            
            if (index === 0) {
                // If there is text before the very first keyword, name it "Introduction" or "Preface"
                fileName = `Part_${String(fileCount).padStart(3, '0')}_Intro.txt`;
            } else {
                // Take the first 30 characters immediately following your keyword
                let titleSnippet = segment.trim().substring(0, 30);
                
                // Clean up formatting: force everything onto one line and replace spaces with underscores
                titleSnippet = titleSnippet.replace(/\s+/g, '_');
                
                // Safe-filtering: strip out characters that Windows/Mac ban in file names (\ / : * ? " < > |)
                titleSnippet = titleSnippet.replace(/[\/\\:\*\?"<>\|]/g, '');
                
                // Construct file name: Keyword + cleaned snippet (e.g., "Chapter_1_The_Start.txt")
                fileName = `${keyword.trim()}_${titleSnippet}.txt`;
            }
            // ---------------------------
            
            // Create preview text for the on-screen checklist text
            const cleanText = segment.trim().replace(/\s+/g, ' ');
            const previewText = cleanText.substring(0, 40) + (cleanText.length > 40 ? '...' : '');
            const displayLabel = `${fileName} — Preview: "${previewText}"`;

            splitFilesMemory.push({
                name: fileName,
                content: fileContent
            });

            const itemDiv = document.createElement('div');
            itemDiv.className = 'chapter-item';

            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.id = `chk_${index}`;
            checkbox.value = fileCount - 1; 
            checkbox.checked = true; 

            const label = document.createElement('label');
            label.htmlFor = `chk_${index}`;
            label.innerText = displayLabel;

            itemDiv.appendChild(checkbox);
            itemDiv.appendChild(label);
            chapterList.appendChild(itemDiv);
        });

        resultsSection.style.display = 'block';
    };

    reader.readAsText(file);
}


function toggleAll(checkedState) {
    const checkboxes = document.querySelectorAll('#chapterList input[type="checkbox"]');
    checkboxes.forEach(cb => cb.checked = checkedState);
}

function getSelectedFiles() {
    const checkboxes = document.querySelectorAll('#chapterList input[type="checkbox"]:checked');
    const selected = [];
    checkboxes.forEach((cb) => {
        const fileIndex = parseInt(cb.value);
        selected.push(splitFilesMemory[fileIndex]);
    });
    return selected;
}

function downloadSelected() {
    const selectedFiles = getSelectedFiles();
    if (selectedFiles.length === 0) {
        alert('Please select at least one file to download.');
        return;
    }

    selectedFiles.forEach((fileData) => {
        const blob = new Blob([fileData.content], { type: 'text/plain;charset=utf-8' });
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = fileData.name;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    });
}

// Custom 100% native client-side TAR encoder (Zero external scripts)
function downloadAsTarArchive() {
    const selectedFiles = getSelectedFiles();
    if (selectedFiles.length === 0) {
        alert('Please select at least one file to bundle.');
        return;
    }

    const encoder = new TextEncoder();
    const chunks = [];

    selectedFiles.forEach(file => {
        const fileData = encoder.encode(file.content);
        
        // Build standard 512-byte header block manually
        const header = new Uint8Array(512);
        
        // 1. Filename (Offset 0, Max length 100)
        for (let i = 0; i < Math.min(file.name.length, 100); i++) {
            header[i] = file.name.charCodeAt(i);
        }
        
        // 2. File Mode (Offset 100, Octal "0000644 \0")
        header.set(encoder.encode("0000644\0"), 100);
        
        // 3. File Size in octal string padded (Offset 124, 11 bytes + space/null)
        const sizeStr = fileData.length.toString(8).padStart(11, '0') + ' ';
        header.set(encoder.encode(sizeStr), 124);
        
        // 4. File modification dummy time (Offset 136)
        header.set(encoder.encode("00000000000\0"), 136);

        // 5. Type Flag '0' for normal file (Offset 156)
        header[156] = 48; 

        // Calculate Header Checksum (Offset 148, size 8)
        header.set(encoder.encode("        "), 148); // Pad with spaces for counting
        let checksum = 0;
        for (let i = 0; i < 512; i++) {
            checksum += header[i];
        }
        const checksumStr = checksum.toString(8).padStart(6, '0') + '\0 ';
        header.set(encoder.encode(checksumStr), 148);

        // Push Header
        chunks.push(header);
        
        // Push File Data Payload
        chunks.push(fileData);
        
        // Pad data to match explicit 512-byte boundary requirements
        const remainder = fileData.length % 512;
        if (remainder !== 0) {
            chunks.push(new Uint8Array(512 - remainder));





}
});

// End of TAR archive marker (Two 512-byte blocks of null characters)

chunks.push(new Uint8Array(1024));

// Compile parts and download
const blob = new Blob(chunks, { type: 'application/x-tar' });
const link = document.createElement('a');
link.href = URL.createObjectURL(blob);
link.download = "Split_Chapters.tar";
document.body.appendChild(link);
link.click();
document.body.removeChild(link);
}
