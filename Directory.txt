    <script>
// Tracking global variables for navigation setup
        let allFilePaths = []; 
        let currentFileIndex = -1;

        async function loadDirectory() {
            const user = document.getElementById('user').value.trim();
            const repo = document.getElementById('repo').value.trim();
            const branch = document.getElementById('branch').value.trim();
            const dirOutput = document.getElementById('dirOutput');
            const copyPathBtn = document.getElementById('copyPathBtn');

            if (!user || !repo || !branch) {
                alert('Please fill out Username, Password, and Path fields.');
                return;
            }

            dirOutput.innerHTML = '<span class="loading">Loading file structure...</span>';
            
            // Reset track data states on new repo initialization
            copyPathBtn.style.display = 'none';
            allFilePaths = [];
            currentFileIndex = -1;

            const url = `https://data.jsdelivr.com/v1/packages/gh/${user}/${repo}@${branch}`;

            try {
                const response = await fetch(url);
                if (!response.ok) throw new Error(`Server returned HTTP ${response.status}`);
                const data = await response.json();

                if (!data.files || data.files.length === 0) {
                    dirOutput.innerHTML = '<span class="error">No files found for this branch.</span>';
                    return;
                }

                dirOutput.innerHTML = '';
                const rootUl = document.createElement('ul');
                rootUl.className = 'root-ul';
                
                buildTreeLayout(data.files, rootUl, "");
                dirOutput.appendChild(rootUl);
                
                // Show path tracking export tool if records exist
                if(allFilePaths.length > 0) {
                    copyPathBtn.style.display = 'block';
                }

            } catch (err) {
                dirOutput.innerHTML = `<span class="error">Failed to fetch directory: ${err.message}</span>`;
            }
        }

        function buildTreeLayout(files, parentElement, currentPath) {
            files.sort((a, b) => (b.type === 'directory') - (a.type === 'directory'));

            files.forEach(item => {
                const li = document.createElement('li');
                const fullPath = currentPath + (currentPath ? '/' : '') + item.name;
                
                if (item.type === 'directory') {
                    const label = document.createElement('span');
                    label.className = 'item-label dir-item';
                    label.innerHTML = `📁 ${item.name}/`;
                    li.appendChild(label);
                    
                    const childUl = document.createElement('ul');
                    li.appendChild(childUl);
                    
                    if (item.files) {
                        buildTreeLayout(item.files, childUl, fullPath);
                    }
                } else {
                    const label = document.createElement('span');
                    label.className = 'item-label file-item';
                    
                    // Assign order indexes to files for linear list navigation matching
                    const localIndex = allFilePaths.length;
                    label.id = `file-${localIndex}`;
                    label.innerHTML = `📄 ${item.name}`;
                    
                    allFilePaths.push(fullPath);
                    
                    label.onclick = () => {
                        currentFileIndex = localIndex;
                        updateActiveHighlight();
                        readFileContent(fullPath);
                    };
                    li.appendChild(label);
                }
                parentElement.appendChild(li);
            });
        }

        // Global storage variable to preserve raw text across view flips
        let activeRawTextPayload = "";

        async function readFileContent(filePath) {
            const user = document.getElementById('user').value.trim();
            const repo = document.getElementById('repo').value.trim();
            const branch = document.getElementById('branch').value.trim();
            const fileOutput = document.getElementById('fileOutput');
            const rawTextOutput = document.getElementById('rawTextOutput');

            fileOutput.className = 'loading';
            fileOutput.innerText = `Streaming raw bytes for: ${filePath}...`;
            rawTextOutput.innerText = "Loading raw string details...";
            activeRawTextPayload = "";

            const url = `https://cdn.jsdelivr.net/gh/${user}/${repo}@${branch}/${filePath}`;

            try {
                const response = await fetch(url);
                if (!response.ok) throw new Error(`HTTP Error ${response.status}`);
                let text = await response.text();
                
                activeRawTextPayload = text; 
                fileOutput.className = '';
                rawTextOutput.innerText = text;
                
                const cleanBase64 = text.replace(/\s/g, '');
                const signature = cleanBase64.substring(0, 8);
                
                let isImage = false;
                let isAudio = false;
                let mimeType = '';

                if (signature.startsWith('iVBOR')) { isImage = true; mimeType = 'image/png'; }
                else if (signature.startsWith('/9j/4')) { isImage = true; mimeType = 'image/jpeg'; }
                else if (signature.startsWith('R0lGO')) { isImage = true; mimeType = 'image/gif'; }
                else if (signature.startsWith('//uQ') || signature.startsWith('SUQz')) { isAudio = true; mimeType = 'audio/mpeg'; }

                if (isImage) {
                    fileOutput.innerHTML = `<img src="data:${mimeType};base64,${cleanBase64}" style="max-width: 100%; max-height: 70vh; border-radius: 4px; display: block; margin: auto;">`;
                } 
                else if (isAudio) {
                    // ADDED: Added tracking IDs and native media listeners
                    fileOutput.innerHTML = `
                        <div style="padding: 20px; text-align: center;">
                            <p style="margin-bottom: 10px; color: #aaa;">🎵 Base64 Audio Playback</p>
                            <audio id="activeAudioPlayer" controls src="data:${mimeType};base64,${cleanBase64}" style="width: 100%; max-width: 400px;"></audio>
                        </div>`;
                    
                    setupMediaAutomationListeners("activeAudioPlayer");
                } 
                else {
                    fileOutput.innerText = text;
                }
                
            } catch (err) {
                fileOutput.className = 'error';
                fileOutput.innerText = `Failed to process text content: ${err.message}`;
                rawTextOutput.innerText = `Error: ${err.message}`;
            }
        }

        // NEW LOGIC: View Flip Mechanics Switching Layout Contexts
        function toggleRawView() {
            const mediaDisplay = document.getElementById('mediaDisplay');
            const rawDisplay = document.getElementById('rawDisplay');
            const isChecked = document.getElementById('rawTextToggle').checked;

            if (isChecked) {
                mediaDisplay.style.display = 'none';
                rawDisplay.style.display = 'block';
            } else {
                mediaDisplay.style.display = 'block';
                rawDisplay.style.display = 'none';
            }
        }

        // UPDATED: Automatically handles standard scrolling, smart shuffles, and loops
        function navigateFile(direction) {
            if (allFilePaths.length === 0) return;
            
            const isShuffleActive = document.getElementById('shuffleToggle').checked;

            if (isShuffleActive && allFilePaths.length > 1) {
                // Pick a completely random index that isn't the current one
                let randomIndex = currentFileIndex;
                while (randomIndex === currentFileIndex) {
                    randomIndex = Math.floor(Math.random() * allFilePaths.length);
                }
                currentFileIndex = randomIndex;
            } else {
                // Standard index progression loops
                currentFileIndex += direction;
                if (currentFileIndex >= allFilePaths.length) currentFileIndex = 0;
                if (currentFileIndex < 0) currentFileIndex = allFilePaths.length - 1;
            }
            
            updateActiveHighlight();
            const targetPath = allFilePaths[currentFileIndex];
            readFileContent(targetPath);
            
            document.getElementById(`file-${currentFileIndex}`)?.scrollIntoView({ block: 'nearest' });
        }

        // NEW AUTOMATION CONNECTOR ENGINE
        function setupMediaAutomationListeners(playerId) {
            const player = document.getElementById(playerId);
            const slider = document.getElementById('globalVolumeSlider');
            const replayBox = document.getElementById('replayToggle');
            if (!player || !slider || !replayBox) return;

            // 1. Instantly set the player volume to match the current slider value
            player.volume = slider.value;

            // 2. Listen for slider movement to dynamically adjust the audio while playing
            slider.oninput = function() {
                player.volume = slider.value;
            };

            // 3. NEW LOOPER LOGIC: Force browser native hardware looping if checked
            player.loop = replayBox.checked;
            
            // Listen for checkbox changes dynamically while the track is active
            replayBox.onchange = function() {
                player.loop = replayBox.checked;
            };

            // Trigger A: If Auto Play is checked, trigger play instantly when loading ends
            const shouldAutoPlay = document.getElementById('autoPlayToggle').checked;
            if (shouldAutoPlay) {
                player.play().catch(err => console.log("Auto-play blocked by browser security. Click interactively first."));
            }

            // Trigger B: If media ends, trigger Auto Next (Native loop bypasses this when active!)
            player.onended = function() {
                const shouldAutoNext = document.getElementById('autoNextToggle').checked;
                // Since player.loop loops natively, onended will ONLY fire if Replay is unchecked
                if (shouldAutoNext) {
                    navigateFile(1); 
                }
            };
        }

        // UPDATED: Automatically grabs raw code background text even if displaying media
        function copyCode() {
            if (!activeRawTextPayload) return;
            navigator.clipboard.writeText(activeRawTextPayload).then(() => {
                const btn = document.getElementById('copyCodeBtn');
                btn.innerText = '✅ Copied Raw!';
                setTimeout(() => btn.innerText = '📋 Copy Content', 1500);
            });
        }

        function updateActiveHighlight() {
            document.querySelectorAll('.file-item').forEach(el => el.classList.remove('active'));
            document.getElementById(`file-${currentFileIndex}`)?.classList.add('active');
        }

        // NEW FEATURE: Clean view frame collapse engine
        function toggleSidebar() {
            document.getElementById('sidebar').classList.toggle('collapsed');
        }

        // NEW FEATURE: Bulk copy text strings
        function copyAllPaths() {
            if (allFilePaths.length === 0) return;
            const pathText = allFilePaths.join('\n');
            navigator.clipboard.writeText(pathText).then(() => {
                const btn = document.getElementById('copyPathBtn');
                const oldText = btn.innerText;
                btn.innerText = '✅ Paths Copied!';
                setTimeout(() => btn.innerText = oldText, 1500);
            });
        }

        // Toggle displaying the volume slider overlay element
        function toggleVolumePopup(event) {
            event.stopPropagation(); // Stops immediate document auto-hiding triggers
            document.getElementById('volumePopup').classList.toggle('show');
        }

        // Global click observer to auto-close the volume menu when clicking elsewhere
        document.addEventListener('click', function(event) {
            const popup = document.getElementById('volumePopup');
            const container = document.querySelector('.volume-container');
            
            // If the element menu is open and click was outside, close it cleanly
            if (popup && popup.classList.contains('show') && !container.contains(event.target)) {
                popup.classList.remove('show');
            }
        });

        // AUTOMATION: Detect window minimize or tab switches to auto-pause media
        document.addEventListener("visibilitychange", function() {
            // Checks if the user has navigated away from the document frame
            if (document.hidden) {
                // Select the active player element if it is currently rendered on screen
                const activePlayer = document.getElementById("activeAudioPlayer");
                if (activePlayer && !activePlayer.paused) {
                    activePlayer.pause();
                    console.log("Tab focus lost: Media paused automatically.");
                }
            }
        });

        // NEW FEATURE: Toggle Zen Mode to hide top input fields and reader headers
        function toggleZenMode() {
            // Find the main controls input panel at the top of the body
            const topControls = document.querySelector('.controls');
            // Find the viewer layout header tracking element
            const viewerHeader = document.getElementById('viewerHeader');
            
            if (topControls && viewerHeader) {
                topControls.classList.toggle('zen-hidden');
                viewerHeader.classList.toggle('zen-hidden');
            }
        }

        // NEW FEATURE: Toggle Theme Engine between Dark and Light Layouts
        function toggleTheme() {
            document.body.classList.toggle('light-theme');
        }
