<script>
    let isEncoder = true;
    
    // The missing piece: This saves your real plaintext memory bank in Encoder mode
    let hiddenPlaintext = ""; 

    const modeTitle = document.getElementById('modeTitle');
    const shiftGroup = document.getElementById('shiftGroup');
    const keyValue = document.getElementById('keyValue');
    const boxLabel = document.getElementById('boxLabel');
    const streamBox = document.getElementById('streamBox');
    const copyBtn = document.getElementById('copyBtn');

    function autoResizeBox() {
        streamBox.style.height = 'auto';
        streamBox.style.height = streamBox.scrollHeight + 'px';
    }

    function getSeedFromString(str) {
        let seed = 0;
        const clean = str.toLowerCase().replace(/[^a-z]/g, '');
        const target = clean.length > 0 ? clean : 'key';
        for (let i = 0; i < target.length; i++) {
            seed = (seed * 31 + target.charCodeAt(i)) % 2147483647;
        }
        return seed;
    }

    function getShiftForPosition(seed, index) {
        let m = 2147483647; 
        let a = 48271;
        let c = 0;
        let val = seed;
        for (let i = 0; i <= index; i++) {
            val = (a * val + c) % m;
        }
        return val % 26;
    }

    function shiftChar(char, shiftAmount, encodeMode) {
        const code = char.charCodeAt(0);
        const finalShift = encodeMode ? shiftAmount : (26 - shiftAmount) % 26;

        if (code >= 65 && code <= 90) {
            return String.fromCharCode(((code - 65 + finalShift) % 26) + 65);
        } else if (code >= 97 && code <= 122) {
            return String.fromCharCode(((code - 97 + finalShift) % 26) + 97);
        }
        return char;
    }

    // Processes a full message string using a non-repeating position seed track
    function processTextBlock(text, seed, encodeMode) {
        let result = '';
        let letterIndex = 0;
        for (let i = 0; i < text.length; i++) {
            if (/[a-zA-Z]/.test(text[i])) {
                const shiftAmount = getShiftForPosition(seed, letterIndex);
                result += shiftChar(text[i], shiftAmount, encodeMode);
                letterIndex++;
            } else {
                result += text[i];
            }
        }
        return result;
    }

    // Refreshes the display text frame layout safely
    function syncDisplay() {
        const seed = getSeedFromString(keyValue.value);
        if (isEncoder) {
            // Re-render the display using the pristine hidden plain text array history
            streamBox.value = processTextBlock(hiddenPlaintext, seed, true);
        } else {
            // Decoder interprets what is directly typed/pasted on screen as cipher content
            hiddenPlaintext = processTextBlock(streamBox.value, seed, false);
            streamBox.value = hiddenPlaintext;
        }
        autoResizeBox();
    }

    // Capture typing input steps safely before it gets corrupted on screen
    streamBox.addEventListener('keypress', (event) => {
        if (event.key === '`' || event.ctrlKey || event.metaKey || event.altKey) return;
        
        if (event.key === 'Enter') {
            event.preventDefault();
            insertDataAtCursor("\n");
            return;
        }

        event.preventDefault();
        insertDataAtCursor(event.key);
    });

    function insertDataAtCursor(newChar) {
        const cursorPosition = streamBox.selectionStart;
        
        if (isEncoder) {
            // Insert your clean typed letter into the secret hidden plain track memory
            hiddenPlaintext = hiddenPlaintext.slice(0, cursorPosition) + newChar + hiddenPlaintext.slice(streamBox.selectionEnd);
            syncDisplay();
        } else {
            // Typing directly inside Decoder mode updates the cipher text strings directly
            const oldCipher = streamBox.value;
            streamBox.value = oldCipher.slice(0, cursorPosition) + newChar + oldCipher.slice(streamBox.selectionEnd);
            syncDisplay();
        }
        
        // Push view selection pointer forward safely
        streamBox.selectionStart = streamBox.selectionEnd = cursorPosition + 1;
        autoResizeBox();
    }

    // input monitoring catches Backspaces, Ctrl+Backspaces, deletions, and cuts cleanly
    streamBox.addEventListener('input', () => {
        const cursor = streamBox.selectionStart;
        if (isEncoder) {
            if (streamBox.value === "") {
                hiddenPlaintext = "";
            } else {
                // Adjust your hidden memory tracker lengths based on structural text box changes
                const lengthDifference = hiddenPlaintext.length - streamBox.value.length;
                if (lengthDifference > 0) {
                    hiddenPlaintext = hiddenPlaintext.slice(0, cursor) + hiddenPlaintext.slice(cursor + lengthDifference);
                }
            }
            syncDisplay();
            streamBox.selectionStart = streamBox.selectionEnd = cursor;
        } else {
            syncDisplay();
            streamBox.selectionStart = streamBox.selectionEnd = cursor;
        }
    });

    // Hidden backtick hotkey logic toggle configurations (`)
    window.addEventListener('keydown', (event) => {
        if (event.key === '`') {
            event.preventDefault();
            isEncoder = !isEncoder;
            const seed = getSeedFromString(keyValue.value);

            if (isEncoder) {
                modeTitle.innerText = "";
                boxLabel.innerText = "Type here...";
                shiftGroup.classList.add('hidden');
                streamBox.value = processTextBlock(hiddenPlaintext, seed, true);
            } else {
                modeTitle.innerText = "Running Decoder Mode";



boxLabel.innerText = "Paste encoded text or type to decode...";
shiftGroup.classList.remove('hidden');
streamBox.value = hiddenPlaintext; // Reveals the clean readable plain track text layout
}
autoResizeBox();
}
});




streamBox.addEventListener('keyup', autoResizeBox);
keyValue.addEventListener('input', syncDisplay);

// Clipboard Copy & Layout Workspace Clear
copyBtn.addEventListener('click', () => {
if (!streamBox.value) return;




navigator.clipboard.writeText(streamBox.value).then(() => {
const originalText = copyBtn.textContent;
copyBtn.textContent = "Copied!";
copyBtn.classList.add('copied');


streamBox.value = '';
hiddenPlaintext = '';
autoResizeBox();





setTimeout(() => {
copyBtn.textContent = originalText;
copyBtn.classList.remove('copied');
}, 1200);
});
});
